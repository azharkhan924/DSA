            sum+=i;
            sqrsum+=i*i;
        }
        int arrsum=0;
        int arrsqrsum=0;
        for(int i=0;i<n;i++){
            arrsum+=nums[i];
            arrsqrsum+=nums[i]*nums[i];
        }
        int diff=sum-arrsum;
        int diffsqr=sqrsum-arrsqrsum;
        int repeating=(diffsqr+diff)/diff;
        int missing=repeating+diff;
        return new int[]{repeating,missing};


    }
}

class Solution {
    public long countInverse(int[] nums) {
        return mergeSort(nums, nums.length);
    }

    public long mergeSort(int[] nums, int n) {
        // Base case
        if (n <= 1) return 0;

        int m1 = n / 2;
        int m2 = n - m1;

        int[] a = new int[m1];
        int[] b = new int[m2];

        for (int i = 0; i < m1; i++) {
            a[i] = nums[i];
        }
        for (int i = 0; i < m2; i++) {
            b[i] = nums[m1 + i];
        }

        long count = 0;
        count += mergeSort(a, m1);
        count += mergeSort(b, m2);
        count += mergeArray(nums, a, b, m1, m2);

        return count;
    }

    public long mergeArray(int[] nums, int[] a, int[] b, int m1, int m2) {
        int[] ans = new int[m1 + m2];
        int i = 0, j = 0, k = 0;
        long count = 0;

        while (i < m1 && j < m2) {
            if (a[i] <= b[j]) {
                ans[k++] = a[i++];
            } else {
                // If a[i] > b[j], all remaining elements in a[i...m1-1] form an inversion with b[j]
                count += (m1 - i);
                ans[k++] = b[j++];
            }
        }

        while (i < m1) ans[k++] = a[i++];
        while (j < m2) ans[k++] = b[j++];

        // Copy back to original array
        for (int p = 0; p < ans.length; p++) {
            nums[p] = ans[p];
        }

        return count;
    }
}




Reverse Pairs

Given an integer array nums, return the number of reverse pairs in the array.

A reverse pair is a pair (i, j) where:

0 <= i < j < nums.length and
nums[i] > 2 * nums[j]

Approach 1: Brute force

Approach 2: 

class Solution {
    public int reversePairs(int[] nums) {
        return mergeSort(nums, nums.length);
    }

    public int mergeSort(int[] nums, int n) {
        if (n <= 1) return 0;

        int m1 = n / 2;
        int m2 = n - m1;

        int[] a = new int[m1];
        int[] b = new int[m2];

        for (int i = 0; i < m1; i++) {
            a[i] = nums[i];
        }
        for (int i = 0; i < m2; i++) {
            b[i] = nums[m1 + i];
        }

        int count = 0;
        count += mergeSort(a, m1);
        count += mergeSort(b, m2);

        // STEP 1: Count reverse pairs before merging
        count += countPairs(a, b, m1, m2);

        // STEP 2: Standard merge to keep array sorted
        mergeArray(nums, a, b, m1, m2);

        return count;
    }

    // Two-pointer counting pass: O(m1 + m2)
    public int countPairs(int[] a, int[] b, int m1, int m2) {
        int count = 0;
        int right = 0;

        for (int i = 0; i < m1; i++) {
            // Note the 2L to avoid integer overflow
            while (right < m2 && (long) a[i] > 2L * b[right]) {
                right++;
            }
            count += right;
        }

        return count;
    }

    // Standard merge (no counting logic here)
    public void mergeArray(int[] nums, int[] a, int[] b, int m1, int m2) {
        int[] ans = new int[m1 + m2];
        int i = 0, j = 0, k = 0;

        while (i < m1 && j < m2) {
            if (a[i] <= b[j]) {
                ans[k++] = a[i++];
            } else {
                ans[k++] = b[j++];
            }
        }

        while (i < m1) ans[k++] = a[i++];
        while (j < m2) ans[k++] = b[j++];

        for (int p = 0; p < ans.length; p++) {
            nums[p] = ans[p];
        }
    }
}





Maximum Product Subarray in an Array

Given an integer array nums, find a subarray that has the largest product, and return the product.

Input: nums = [2,3,-2,4]
Output: 6

Approach 1: nested loop  brute force
Approach 2: Optimal approach 

1.for positve element multiply all
2. for even -ve multiply all
3. odd -ve any one -ve no. is breaking pointer 
4. if 0 is there than zero is also breaking point.

therefore calculate prefix and suffix product both and during this process mantain a max variable for identifying the max propduct value 
if prefix or sufix becomes zero set them to 1


class Solution {
        int max = Integer.MIN_VALUE;
        int prefix=1,suffix=1;
        int n =nums.length;
        for(int i=0;i<n;i++){
            prefix*=nums[i];
            suffix*=nums[n-1-i];
            max = Math.max(max,Math.max(prefix,suffix));
            if(prefix==0)prefix=1;
            if(suffix==0)suffix=1;

        }
        return max;
    }
